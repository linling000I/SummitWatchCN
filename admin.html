关于本项目

网络黑名单（Network Blacklist）是 PEAK 游戏的跨玩家共享作弊封禁名单。玩家用 Steam 登录即可举报作弊者、对误封提交申诉；管理员审核后，所有装了对应模组的客户端会自动同步并踢出被封禁玩家。

本名单对其他反作弊 / 工具模组开放只读接口：任何 BepInEx 模组都可以通过反射调用 SummitWatchCN.API 查询封禁状态、读取名单快照、触发同步，无需重复实现网络层。

开发者 API（其他反作弊模组可调用） 
var asm = AppDomain.CurrentDomain.GetAssemblies()
    .FirstOrDefault(a => a.GetName().Name == "SummitWatchCN");
var api = asm?.GetType("SummitWatchCN.API");

api.GetMethod("IsBannedAnywhere").Invoke(null, new object[]{ steamId, playerName });
api.GetMethod("IsNetworkBanned").Invoke(null, new object[]{ steamId });
api.GetMethod("GetNetworkEntries").Invoke(null, null);
api.GetMethod("GetLocalEntries").Invoke(null, null);
api.GetMethod("RefreshNetworkNow").Invoke(null, null);
api.GetProperty("IsReady").GetValue(null);
api.GetProperty("HasPeakSentinel").GetValue(null);
